syntax · ศัพท์ · pattern · โลกจริง
ศัพท์ที่ต้องเข้าใจก่อนเขียน script ทุกคำอธิบายแบบเข้าใจง่าย
data typeObject
ข้อมูลที่มีหลาย field ห่อด้วย { } — เหมือนกล่องที่มีของข้างใน
{ "id": 1, "name": "Ann", "isActive": "Y" }
data typeArray
รายการข้อมูลหลายอัน ห่อด้วย [ ] เริ่มนับจาก index 0 เสมอ
[ {"id":1}, {"id":2} ] ← [0]=อันแรก [1]=อันสอง
conceptProperty / Field / Key
ชื่อของข้อมูลใน object ใช้แทนกันได้ทั้ง 3 คำ ความหมายเดียวกัน
{ "id": 1 } ← "id" คือ property · 1 คือ value
conceptNested Object
object อยู่ข้างใน object อีกทีนึง เข้าถึงด้วยจุด (.) ต่อกัน
{ "address": { "city": "Bangkok" } } → body.address.city ได้ "Bangkok"
testingAssertion
การยืนยันว่าค่าต้องเป็นแบบนี้ ถ้าไม่ตรง test จะ fail อัตโนมัติ
pm.expect(body.id).to.equal(1) ← assert ว่า id ต้องเป็น 1
testingHappy Path
ส่งข้อมูลถูกต้องครบ → คาดว่า API ทำงานสำเร็จ ได้ 200/201
POST user ด้วย email + password ถูก → ได้ 200 + token
testingNegative Test
ส่งข้อมูลผิด/ไม่ครบ → เช็คว่า API จัดการ error ได้ถูกต้อง
POST โดยไม่ส่ง required field → คาดว่าได้ 400
specM / O (Mandatory / Optional)
M = ต้องมีเสมอ ถ้าไม่มี = bug · O = มีหรือไม่มีก็ได้ ต้องเช็คใน if
M → pm.expect(body).to.have.property('id') O → if (body.minStock !== undefined) { เช็คต่อ }
specSV / MV (Single / Multiple Value)
SV = ค่าเดียว · MV = หลายค่า เป็น array ต้องเช็คให้ต่างกัน
MV → pm.expect(body.specs).to.be.an('array')
conceptChain Request
เอาค่าจาก response หนึ่ง ไปใช้ใน request ถัดไปอัตโนมัติ ไม่ต้อง copy ด้วยมือ
POST → ได้ id → เก็บใน variable → GET /items/{{id}}
conceptSchema Validation
เช็คว่า response มีครบทุก field และ type ถูกต้องตาม spec
spec บอก id เป็น String → เช็คว่า body.id เป็น 'string' ไม่ใช่ number
ทุกครั้งที่เขียน script ต้องเริ่มจาก 2 บรรทัดนี้ก่อนเสมอ
START HEREconst body = pm.response.json() // แปลง response ก่อน console.log(body) // ดูก่อนเขียน assert เสมอ
เข้าถึง Object
// response: { "id": 1, "name": "Ann", "isActive": "Y" } body.id // ได้ 1 body.name // ได้ "Ann" body.isActive // ได้ "Y"
เข้าถึง Nested Object
// response: { "assetGroupObj": { "cfgName": "Main" } } body.assetGroupObj // ได้ { "cfgName": "Main" } body.assetGroupObj.cfgName // ได้ "Main"
เข้าถึง Array
// response: [ {"id":1,"name":"A"}, {"id":2,"name":"B"} ] body[0] // อันแรก index=0 body[0].id // ได้ 1 body[1].name // ได้ "B" body.length // จำนวนทั้งหมด body[body.length-1] // อันสุดท้าย
Array ซ้อนใน Object
// { "resultData": [ { "specs": [ {"specName":"CPU"} ] } ] } body.resultData[0].specs[0].specName // ได้ "CPU"
💡 console.log(body) ก่อนเสมอ → เห็นค่า → แล้วค่อยเขียน assert
โครงสร้างหลัก
pm.test("ชื่อ test", function () { pm.expect(ค่าที่มี).to.___(ค่าที่คาดหวัง) })
เช็คค่าตรงๆ
pm.expect(body.resultCode).to.equal("20000") pm.expect(body.resultCode).to.not.equal("99999") pm.expect(body).to.deep.equal({})
เช็คชนิดข้อมูล (type)
pm.expect(body.id).to.be.a('string') pm.expect(body.count).to.be.a('number') pm.expect(body.specs).to.be.an('array') pm.expect(body.data).to.be.an('object') pm.expect(body.flag).to.be.true pm.expect(body.deleted).to.be.null
เช็คว่ามี field อยู่
pm.expect(body).to.have.property('id') pm.expect(body).to.have.all.keys(['id', 'name', 'isActive'])
เช็คตัวเลข / string / array
pm.expect(body.price).to.be.above(0) pm.expect(body.age).to.be.below(100) pm.expect(body.email).to.include('@') pm.expect(body.isActive).to.be.oneOf(['Y', 'N']) pm.expect(pm.response.responseTime).to.be.below(2000)
💡 เพิ่ม .not. หน้าอะไรก็ได้ เช่น .to.not.be.null = ต้องไม่เป็น null

⚠️ Optional field (O) ต้องเช็คใน if ก่อนเสมอ ไม่งั้นจะ fail ตอน field ไม่มี
// ❌ ผิด pm.expect(body.minStock).to.be.a('number') // ✅ ถูก if (body.minStock !== undefined) { pm.expect(body.minStock).to.be.a('number') }

const token = pm.environment.get("token") pm.request.headers.add({ key: "Authorization", value: `Bearer ${token}` }) const r = Math.floor(Math.random() * 99999) pm.collectionVariables.set("email", `test${r}@mail.com`)

4 pattern นี้ครอบคลุม 90% ของงานจริง จำ pattern ได้ก็ประยุกต์ได้ทันที
1
เช็ค M/O + SV/MV ตาม spec
ใช้ทุกครั้งที่มี spec งาน เช็ค Mandatory ก่อน แล้วค่อยเช็ค Optional ใน if
const body = pm.response.json() const first = body.resultData[0] // ดึง object อันแรกออกมาใช้ pm.test("M fields ต้องครบ", function () { pm.expect(first).to.have.property('assetTypeId') // M pm.expect(first).to.have.property('isActive') // M }) if (first.minStock !== undefined) { // O: เช็คก่อนว่ามีค่าไหม pm.test("minStock เป็น number", function () { pm.expect(first.minStock).to.be.a('number') }) } pm.test("specs เป็น array (MV)", function () { pm.expect(first.specs).to.be.an('array') // MV ต้องเป็น array if (first.specs.length > 0) { // ถ้ามีข้อมูลค่อยเช็คต่อ pm.expect(first.specs[0]).to.have.property('specName') } })
📦body.resultData[0]ดึง object อันแรกออกมาเก็บไว้ใน first ไม่ต้องพิมพ์ซ้ำทุกบรรทัด
✅have.property('x')เช็คว่ามี field ชื่อ x อยู่ใน object — ใช้กับ M field ที่ต้องมีเสมอ
❓if (x !== undefined)เช็คก่อนว่า field มีค่าไหม ถ้าไม่มีจะข้ามไป ไม่ fail — ใช้กับ O field
📋be.an('array')เช็คว่าเป็น array — ใช้กับ MV field ที่มีหลายค่า
🔢specs.length > 0เช็คว่า array ไม่ว่างก่อน แล้วค่อยเข้าถึง [0] ไม่งั้นจะ error
2
POST แล้วบันทึก id สำหรับ chain
หลัง POST สำเร็จ บันทึก id ไว้ใช้ใน GET หรือ DELETE อัตโนมัติ ไม่ต้อง copy มือ
const body = pm.response.json() pm.test("สร้างสำเร็จ", function () { pm.response.to.have.status(200) // HTTP status ถูก pm.expect(body.resultCode).to.equal("20000") // business code ถูก pm.expect(body.resultData).to.have.property('assetTypeId') // มี id กลับมา }) // บันทึก id ไว้ใช้ใน request ถัดไป (วางนอก pm.test) pm.collectionVariables.set("assetTypeId", body.resultData.assetTypeId) console.log("saved id:", body.resultData.assetTypeId)
🔍have.status(200)เช็ค HTTP status — ต้องผ่านก่อนถึงจะเช็ค business logic ต่อได้
💼resultCode "20000"เช็ค business status code ของระบบ ต่างจาก HTTP status — ต้องเช็คทั้งคู่
💾collectionVariables.setบันทึก id ไว้ใน memory — request ถัดไปใช้ {{assetTypeId}} ใน URL ได้เลย
📍วางนอก pm.testการบันทึกค่าไม่ใช่ assertion จึงวางข้างนอก pm.test ไม่ต้องห่อไว้ข้างใน
3
Negative test ครบ 3 case
ทำเป็น request แยกกัน แต่ละ request เช็ค 1 error case ไม่รวมกัน
// Request A — ไม่ส่ง required field pm.test("ขาด required field → 400", function () { pm.response.to.have.status(400) // ต้องได้ 400 ไม่ใช่ 200 pm.expect(body.resultCode).to.not.equal("20000") // resultCode ต้องไม่ success }) // Request B — ส่งข้อมูลซ้ำ 2 ครั้ง pm.test("duplicate → 409", function () { pm.response.to.have.status(409) // conflict เพราะซ้ำ }) // Request C — ลบ Authorization header ออก pm.test("no token → 401", function () { pm.response.to.have.status(401) // unauthorized })
🚫to.not.equal("20000")เช็คว่าไม่ success — เพราะ error case ต้องไม่ได้ resultCode 20000 กลับมา
📨request แยกกันแต่ละ error case ทำเป็น request แยก เพราะ body ที่ส่งต่างกัน ไม่ควนรวมกัน
🗝️ลบ headerทดสอบ 401 ด้วยการลบ Authorization header ออกจาก request นั้นๆ
4
วน loop เช็คทุก item ใน array
ใช้เมื่อ resultData มีหลายอัน อยากเช็คให้ครบทุก item ไม่ใช่แค่ [0]
body.resultData.forEach(function (item, index) { // item = object แต่ละอัน, index = ลำดับที่ (0,1,2...) pm.test(`item[${index}] มี assetTypeId`, function () { pm.expect(item).to.have.property('assetTypeId') // เช็คทุก item ว่ามี field นี้ ไม่ใช่แค่อันแรก }) pm.test(`item[${index}] isActive เป็น Y หรือ N`, function () { pm.expect(item.isActive).to.be.oneOf(['Y', 'N']) // ค่าต้องเป็นแค่ Y หรือ N เท่านั้น ค่าอื่นถือว่า fail }) })
🔄forEach(item, index)วนทุก item ใน array อัตโนมัติ — item คือข้อมูล index คือลำดับที่
🏷️`item[${index}]`ชื่อ test จะแสดงเป็น item[0], item[1]... ทำให้รู้ทันทีว่า item ไหน fail
🎯be.oneOf([...])เช็คว่าค่าต้องเป็นหนึ่งในนี้เท่านั้น — เหมาะกับ field ที่มีค่าจำกัด เช่น Y/N
⚡ทำไมไม่ใช้แค่ [0]ถ้าเช็คแค่ [0] อาจพลาด bug ที่อยู่ใน item ลำดับอื่น loop จะจับได้ทุกอัน
💡 4 pattern นี้ครอบคลุม 90% ของงานจริง จำ pattern ได้ก็ประยุกต์ได้เลย ไม่ต้องจำ syntax ทุกตัว
